use IEEE.STD_LOGIC_1164.ALL;

architecture rtl of inst_mem is
  type rom_t is array (0 to 255) of std_logic_vector(31 downto 0);
  constant ROM : rom_t := (
    0 => x"00500293",  -- addi x5, x0, 5
    1 => x"00300313",  -- addi x6, x0, 3

    others => (others => '0') );
begin

end rtl;
